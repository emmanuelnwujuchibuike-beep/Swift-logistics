Database Setup SQL
Run this SQL in your Supabase dashboard:
Go to supabase.com → your project → SQL Editor → New query, paste the code below, and click Run. This creates all required tables and is safe to re-run at any time. 
-- ══════════════════════════════════════════════
-- Swift Freight Logistics — Database Setup
-- Run in Supabase SQL Editor (safe to re-run)
-- ══════════════════════════════════════════════

-- ── Site Settings ────────────────────────────
CREATE TABLE IF NOT EXISTS public.site_settings (
  key        text PRIMARY KEY,
  value      text,
  updated_at timestamptz DEFAULT now()
);
ALTER TABLE public.site_settings ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Public can read settings" ON public.site_settings;
CREATE POLICY "Public can read settings"
  ON public.site_settings FOR SELECT USING (true);
INSERT INTO public.site_settings (key, value)
  VALUES ('ticker', '') ON CONFLICT (key) DO NOTHING;

-- ── Shipments table ──────────────────────────
CREATE TABLE IF NOT EXISTS public.shipments (
  id               BIGSERIAL PRIMARY KEY,
  tracking_id      TEXT UNIQUE NOT NULL,
  customer_email   TEXT,
  name             TEXT,
  senders_name     TEXT,
  status           TEXT DEFAULT 'Pending',
  current_location TEXT,
  destination      TEXT,
  origin           TEXT,
  pickup_date      TEXT,
  eta              TEXT,
  package_details  TEXT,
  service_type     TEXT DEFAULT 'Standard Freight',
  priority         TEXT DEFAULT 'Standard',
  payment_status   TEXT DEFAULT 'not-required',
  amount_due       TEXT,
  btc_address      TEXT,
  usdt_address     TEXT,
  bank_name        TEXT,
  account_name     TEXT,
  bank_number      TEXT,
  routing_number   TEXT,
  paypal_email     TEXT,
  cashapp_tag      TEXT,
  zelle_id         TEXT,
  western_union_info TEXT,
  venmo_tag        TEXT,
  moneygram_info   TEXT,
  amazon_gc_info   TEXT,
  google_gc_info   TEXT,
  apple_gc_info    TEXT,
  vanilla_gc_info  TEXT,
  ebay_gc_info     TEXT,
  notes            TEXT,
  step1_color      TEXT DEFAULT 'text-blue-500',
  step2_name       TEXT, step2_location TEXT, step2_color TEXT DEFAULT 'text-blue-500',
  step3_name       TEXT, step3_location TEXT, step3_color TEXT DEFAULT 'text-blue-500',
  step4_name       TEXT, step4_location TEXT, step4_color TEXT DEFAULT 'text-green-500',
  created_at       TIMESTAMPTZ DEFAULT NOW(),
  updated_at       TIMESTAMPTZ DEFAULT NOW()
);

-- ── Add new columns to existing tables (safe) ─
ALTER TABLE public.shipments
  ADD COLUMN IF NOT EXISTS routing_number     TEXT,
  ADD COLUMN IF NOT EXISTS paypal_email       TEXT,
  ADD COLUMN IF NOT EXISTS cashapp_tag        TEXT,
  ADD COLUMN IF NOT EXISTS zelle_id           TEXT,
  ADD COLUMN IF NOT EXISTS western_union_info TEXT,
  ADD COLUMN IF NOT EXISTS venmo_tag          TEXT,
  ADD COLUMN IF NOT EXISTS moneygram_info     TEXT,
  ADD COLUMN IF NOT EXISTS amazon_gc_info     TEXT,
  ADD COLUMN IF NOT EXISTS google_gc_info     TEXT,
  ADD COLUMN IF NOT EXISTS apple_gc_info      TEXT,
  ADD COLUMN IF NOT EXISTS vanilla_gc_info    TEXT,
  ADD COLUMN IF NOT EXISTS ebay_gc_info       TEXT;

-- ── Row Level Security ───────────────────────
ALTER TABLE public.shipments ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Public read shipments" ON public.shipments;
CREATE POLICY "Public read shipments"
  ON public.shipments FOR SELECT USING (true);
DROP POLICY IF EXISTS "Service role full access" ON public.shipments;
CREATE POLICY "Service role full access"
  ON public.shipments USING (auth.role() = 'service_role');

-- ── Public view (rebuild every time) ─────────
DROP VIEW IF EXISTS public.shipments_public;
CREATE VIEW public.shipments_public AS
  SELECT
    id, created_at, updated_at,
    tracking_id, name, senders_name, status,
    current_location, destination, origin,
    pickup_date, eta, package_details, service_type, priority,
    payment_status, amount_due,
    btc_address, usdt_address,
    bank_name, account_name, bank_number, routing_number,
    paypal_email, cashapp_tag, zelle_id, western_union_info,
    venmo_tag, moneygram_info,
    amazon_gc_info, google_gc_info, apple_gc_info,
    vanilla_gc_info, ebay_gc_info,
    notes,
    step1_color,
    step2_name, step2_location, step2_color,
    step3_name, step3_location, step3_color,
    step4_name, step4_location, step4_color
  FROM public.shipments;

GRANT SELECT ON public.shipments_public TO anon, authenticated;

-- ── Quote requests ───────────────────────────
CREATE TABLE IF NOT EXISTS public.quote_requests (
  id           BIGSERIAL PRIMARY KEY,
  reference    TEXT UNIQUE NOT NULL,
  full_name    TEXT, email TEXT, phone TEXT,
  origin       TEXT, destination TEXT,
  cargo_type   TEXT, weight TEXT, message TEXT,
  created_at   TIMESTAMPTZ DEFAULT NOW()
);
ALTER TABLE public.quote_requests ENABLE ROW LEVEL SECURITY;

-- ── Timestamp trigger ────────────────────────
CREATE OR REPLACE FUNCTION update_updated_at()
RETURNS TRIGGER AS $$
BEGIN NEW.updated_at = NOW(); RETURN NEW; END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS set_updated_at ON public.shipments;
CREATE TRIGGER set_updated_at
  BEFORE UPDATE ON public.shipments
  FOR EACH ROW EXECUTE FUNCTION update_updated_at();

-- ── Indexes ───────────────────────────────────
CREATE INDEX IF NOT EXISTS shipments_tracking_id_idx ON public.shipments (tracking_id);
CREATE INDEX IF NOT EXISTS shipments_email_idx       ON public.shipments (customer_email);

-- ── Live Chat ─────────────────────────────────────────────────
CREATE TABLE IF NOT EXISTS public.chat_messages (
  id            BIGSERIAL PRIMARY KEY,
  session_id    TEXT NOT NULL,
  role          TEXT NOT NULL CHECK (role IN ('visitor','agent')),
  content       TEXT NOT NULL,
  visitor_name  TEXT,
  visitor_email TEXT,
  page_url      TEXT,
  created_at    TIMESTAMPTZ DEFAULT NOW()
);
-- ADD COLUMN IF NOT EXISTS is safe to re-run on existing tables
ALTER TABLE public.chat_messages ADD COLUMN IF NOT EXISTS visitor_email TEXT;
ALTER TABLE public.chat_messages ADD COLUMN IF NOT EXISTS page_url TEXT;
ALTER TABLE public.chat_messages ADD COLUMN IF NOT EXISTS image_url TEXT;
CREATE INDEX IF NOT EXISTS chat_msg_session_idx ON public.chat_messages (session_id, created_at);
ALTER TABLE public.chat_messages ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "chat_open_access" ON public.chat_messages;
CREATE POLICY "chat_open_access" ON public.chat_messages FOR ALL USING (true) WITH CHECK (true);
GRANT ALL ON public.chat_messages TO anon, authenticated;
GRANT USAGE, SELECT ON SEQUENCE public.chat_messages_id_seq TO anon, authenticated;

-- ── Chat Sessions (status tracking) ──────────────────────────
CREATE TABLE IF NOT EXISTS public.chat_sessions (
  session_id    TEXT PRIMARY KEY,
  status        TEXT DEFAULT 'open' CHECK (status IN ('open','resolved')),
  visitor_name  TEXT,
  visitor_email TEXT,
  created_at    TIMESTAMPTZ DEFAULT NOW(),
  updated_at    TIMESTAMPTZ DEFAULT NOW()
);
ALTER TABLE public.chat_sessions ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "chat_sessions_access" ON public.chat_sessions;
CREATE POLICY "chat_sessions_access" ON public.chat_sessions FOR ALL USING (true) WITH CHECK (true);
GRANT ALL ON public.chat_sessions TO anon, authenticated;

-- ── Enable Supabase Realtime on chat tables ───────────────────
-- REPLICA IDENTITY FULL is idempotent (safe to re-run).
ALTER TABLE public.chat_messages REPLICA IDENTITY FULL;
ALTER TABLE public.chat_sessions  REPLICA IDENTITY FULL;

-- ADD TABLE is NOT idempotent, so we guard with a DO block.
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_publication_tables
    WHERE pubname = 'supabase_realtime' AND tablename = 'chat_messages'
  ) THEN
    ALTER PUBLICATION supabase_realtime ADD TABLE public.chat_messages;
  END IF;

  IF NOT EXISTS (
    SELECT 1 FROM pg_publication_tables
    WHERE pubname = 'supabase_realtime' AND tablename = 'chat_sessions'
  ) THEN
    ALTER PUBLICATION supabase_realtime ADD TABLE public.chat_sessions;
  END IF;
END $$;

-- ── Supabase Storage: chat-images bucket ──────────────────────
INSERT INTO storage.buckets (id, name, public)
VALUES ('chat-images', 'chat-images', true)
ON CONFLICT (id) DO NOTHING;

DROP POLICY IF EXISTS "chat_images_upload" ON storage.objects;
CREATE POLICY "chat_images_upload" ON storage.objects
  FOR INSERT TO anon, authenticated
  WITH CHECK (bucket_id = 'chat-images');

DROP POLICY IF EXISTS "chat_images_read" ON storage.objects;
CREATE POLICY "chat_images_read" ON storage.objects
  FOR SELECT TO anon, authenticated
  USING (bucket_id = 'chat-images');
Security & Credentials
All credentials are stored securely in Supabase Edge Function Secrets — never in the page source code.

To set or change credentials:
Supabase Dashboard → your project → Edge Functions → Secrets

Required variables:
  • ADMIN_PASSWORD — login password for this panel
  • ADMIN_API_SECRET — secret sent to edge functions
  • TOKEN_SECRET — random string for signing sessions (any long random text)

After updating env vars, redeploy the site for changes to take effect.
